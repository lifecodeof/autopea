// === Apply a layer effect (Solid Fill) with specific color to the active layer ===

/* Params
 * param_layer: Layer
 * param_red: number
 * param_green: number
 * param_blue: number
 */

// Select param_layer
if (app.activeDocument.activeLayer != param_layer) {
  app.activeDocument.activeLayer = param_layer;
}

// Main descriptor for the 'set' action
var desc = new ActionDescriptor();

// Create a reference to the layer effect property of the currently selected layer
var ref = new ActionReference();
ref.putProperty(charIDToTypeID("Prpr"), charIDToTypeID("Lefx")); // "Prpr" = property, "Lefx" = Layer Effects
ref.putEnumerated(charIDToTypeID("Lyr "), charIDToTypeID("Ordn"), charIDToTypeID("Trgt"));
// "Lyr " = Layer
// "Ordn" = Order (kind of selection criteria)
// "Trgt" = Target (i.e. the currently selected layer)
desc.putReference(charIDToTypeID("null"), ref); // Assign the target reference to the action

// === Define the new layer effect ===
var effectDesc = new ActionDescriptor();

// Scale the layer style effects (e.g. shadows, strokes) — 100% = normal size
effectDesc.putUnitDouble(charIDToTypeID("Scl "), charIDToTypeID("#Prc"), 100.0); 
// "Scl " = Scale
// "#Prc" = Percent

// === Define the Solid Fill (SoFi) layer style ===
var soFiDesc = new ActionDescriptor();

// Enable the Solid Fill
soFiDesc.putBoolean(charIDToTypeID("enab"), true); // "enab" = enabled

// Set blend mode to Normal
soFiDesc.putEnumerated(charIDToTypeID("Md  "), charIDToTypeID("BlnM"), charIDToTypeID("Nrml"));
// "Md  " = Blend Mode
// "BlnM" = Blend Mode enum type
// "Nrml" = Normal

// Set opacity to 100%
soFiDesc.putUnitDouble(charIDToTypeID("Opct"), charIDToTypeID("#Prc"), 100.0); 
// "Opct" = Opacity

// === Define the fill color: RGB(18, 52, 86) ===
var colorDesc = new ActionDescriptor();
colorDesc.putDouble(charIDToTypeID("Rd  "), param_red);  // Red
colorDesc.putDouble(charIDToTypeID("Grn "), param_green);  // Green
colorDesc.putDouble(charIDToTypeID("Bl  "), param_blue);  // Blue

// Assign color to the solid fill
soFiDesc.putObject(charIDToTypeID("Clr "), charIDToTypeID("RGBC"), colorDesc); 
// "Clr " = Color
// "RGBC" = RGB Color type

// Attach the solid fill style to the layer effects
effectDesc.putObject(charIDToTypeID("SoFi"), charIDToTypeID("SoFi"), soFiDesc); 
// "SoFi" = Solid Fill

// Assign the effect description to the final descriptor
desc.putObject(charIDToTypeID("T   "), charIDToTypeID("Lefx"), effectDesc); 
// "T   " = Target value (the "what to set it to")

// === Finally, execute the action ===
executeAction(charIDToTypeID("setd"), desc, DialogModes.NO); 
// "setd" = Set action (actually performs the change)
// DialogModes.NO = Don't show any dialog boxes
